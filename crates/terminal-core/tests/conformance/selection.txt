# Selection (TERM-17): what a drag, a double-click, a triple-click, a block and Select All
# select, the text copying gives, and when a selection ends. xterm is no model here: what a
# selection takes follows DESIGN.md §3 *Selection*, and it ends when a row it covers changes,
# as in kitty and Alacritty.

## A drag selects from one boundary between cells to another
size 6x2
send abcdef\r\nghijkl
select characters 1;2
extend 2;3
selected
|.*****|
|**....|
copy bcdef\ngh

## The right half of a cell is the boundary after it
size 6x1
send abcdef
select characters 1;2+
extend 1;4+
selected
|..**..|
copy cd

## A drag back selects the same cells
size 6x1
send abcdef
select characters 1;5
extend 1;2
copy bcd

## A click selects nothing
size 6x1
send abcdef
select characters 1;2
copy
selected
|......|

## A press ends the selection there was
size 6x1
send abcdef
select characters 1;1
extend 1;6+
copy abcdef
select characters 1;3
copy

## Unselect ends it
size 6x1
send abcdef
select characters 1;1
extend 1;6+
unselect
copy

## A wide character is selected whole, from either of its cells
size 6x1
send a中b
select characters 1;3
extend 1;5
selected
|.***..|
copy 中b

## Copying joins the rows of a line that wraps, and drops the spaces at the end of each line
size 4x3
send abcdefg\r\nhi\x20\x20
screen
|abcd|>
|efg |
|hi  |
select characters 1;1
extend 3;4+
copy abcdefg\nhi

## The spaces where a line wraps are part of it
size 4x2
send ab\x20\x20cd
screen
|ab  |>
|cd  |
select characters 1;1
extend 2;2+
copy ab\x20\x20cd

## The spaces at the end of a selection that stops inside a line are dropped too
size 6x1
send ab\x20\x20cd
select characters 1;1
extend 1;4+
copy ab

## The cell a wide character left when it went on to the next row is not copied
# reflow.txt has the same cell: it is not part of the line.
size 4x2
send abc中
screen
|abc |»
|中  |
select characters 1;1
extend 2;2+
copy abc中

## A blank line between two is copied as an empty line
size 4x3
send a\r\n\r\nb
select characters 1;1
extend 3;4+
copy a\n\nb

## The marks joined to a character, ZWJ and the variation selectors are copied as they came
size 8x1
send e\xcc\x81 \xe2\x9d\xa4\xef\xb8\x8f \xf0\x9f\x91\xa9\xe2\x80\x8d\xf0\x9f\x92\xbb
select lines 1;1
copy e\xcc\x81 \xe2\x9d\xa4\xef\xb8\x8f \xf0\x9f\x91\xa9\xe2\x80\x8d\xf0\x9f\x92\xbb

## A double-click selects the word under it
size 12x1
send src/a.rs:12
select words 1;3
selected
|********....|
copy src/a.rs

## A double-click on a separator or a blank selects it alone
# Blanks alone copy nothing, so the clipboard keeps what it had.
size 12x1
send src/a.rs:12
select words 1;9
copy :
select words 1;12
selected
|...........*|
copy

## A wide blank ends a word as a narrow one does
# U+3000, the ideographic space, takes two cells. It is a character the program wrote, so copying
# keeps it: only the blanks at the end of a line go.
size 8x1
send a\xe3\x80\x80bc
select words 1;4
selected
|...**...|
copy bc
select words 1;3
selected
|.**.....|
copy \xe3\x80\x80

## A wide character is part of a word
size 8x1
send ab中cd e
select words 1;1
copy ab中cd
select words 1;4
copy ab中cd

## A drag after a double-click takes whole words, and keeps the first one
size 13x1
send one two three
select words 1;6
copy two
extend 1;10
copy two three
extend 1;1
copy one two

## A drag after a double-click past the top or the bottom takes the words there
size 16x3
send alpha beta gamma\r\nxx\r\nyy
select words 3;1
extend -5;12
copy alpha beta gamma\nxx\nyy
select words 1;8
extend 9;1
copy beta gamma\nxx\nyy

## A block takes the columns between its corners on each row
size 5x3
send abcde\r\nfghij\r\nklmno
select block 1;2
extend 3;3+
selected
|.**..|
|.**..|
|.**..|
copy bc\ngh\nlm

## A block from the bottom right to the top left is the same block
size 5x3
send abcde\r\nfghij\r\nklmno
select block 3;3+
extend 1;2
copy bc\ngh\nlm

## A block of no columns selects nothing
size 5x3
send abcde\r\nfghij\r\nklmno
select block 1;2
extend 3;2
copy

## A block does not join the rows a line wraps over
size 3x2
send abcdef
select block 1;1
extend 2;2+
copy ab\nde

## Rows dropped past the scrollback's limit are cut from the selection's start
size 4x2
send a\r\nb\r\nc\r\nd
select characters -1;1
extend 2;1+
copy a\nb\nc\nd
limit 1
copy b\nc\nd
limit 0
copy c\nd

## A selection whose rows were all dropped past the limit selects nothing
size 4x2
send a\r\nb\r\nc
select characters 0;1
extend 0;1+
copy a
limit 0
copy
