# Colors programs ask for and set (TERM-12). The theme's colors answer, and a color a program
# set answers and is drawn in their place until it is given back. ctlseqs: OSC 4, OSC 10 to 19,
# OSC 104 and OSC 110 to 119. The theme here is the suite's own (conformance.rs): the
# foreground #c0c0c0, the background #101010, the cursor #f0f000, and color n of the palette n,
# n and n. Each answer has a component's 8 bits twice, as xterm's on a display of 24-bit color.

## OSC 10, 11 and 12 ask for the foreground, the background and the cursor
# The answer ends as the question did.
size 10x1
send \e]10;?\a\e]11;?\e\\\e]12;?\a
reply \e]10;rgb:c0c0/c0c0/c0c0\a
reply \e]11;rgb:1010/1010/1010\e\\
reply \e]12;rgb:f0f0/f0f0/0000\a

## OSC 4 asks for colors of the palette, a reply for each
size 10x1
send \e]4;1;?;255;?\e\\
reply \e]4;1;rgb:0101/0101/0101\e\\
reply \e]4;255;rgb:ffff/ffff/ffff\e\\

## Each parameter of OSC 10 is the next dynamic color
# ctlseqs: "Each successive parameter changes the next color in the list." 13 to 19 are not
# the theme's, and are not answered.
size 10x1
send \e]10;?;?;?;?\a\e]11;?;?\a
reply \e]10;rgb:c0c0/c0c0/c0c0\a
reply \e]11;rgb:1010/1010/1010\a
reply \e]12;rgb:f0f0/f0f0/0000\a
reply \e]11;rgb:1010/1010/1010\a
reply \e]12;rgb:f0f0/f0f0/0000\a

## The answer comes in order with the other replies
# Programs ask for DA1 after a color, and take DA1 coming first as no answer.
size 10x1
send \e]11;?\a\e[c\e]4;0;?\a
reply \e]11;rgb:1010/1010/1010\a
reply \e[?62;22c
reply \e]4;0;rgb:0000/0000/0000\a

## A color a program set is the answer
# 8 bits of each component are kept: the high ones of the 16 XParseColor reads.
size 10x1
send \e]11;rgb:1234/5678/9abc\a\e]11;?\a\e]4;1;#ff8000\a\e]4;1;?\a\e]10;rgb:f/80/abc\a\e]10;?\a
send \e]12;RGB:00/ff/00\a\e]12;?\a
reply \e]11;rgb:1212/5656/9a9a\a
reply \e]4;1;rgb:ffff/8080/0000\a
reply \e]10;rgb:ffff/8080/abab\a
reply \e]12;rgb:0000/ffff/0000\a

## rgb: scales its digits, and # takes them as the high bits
# XParseColor: rgb:1/2/3 is 1111/2222/3333, and #123 is 1000/2000/3000.
size 10x1
send \e]4;2;#123\a\e]4;3;rgb:1/2/3\a\e]4;4;#123456789\a\e]4;5;#123456789abc\a\e]4;2;?;3;?;4;?;5;?\a
reply \e]4;2;rgb:1010/2020/3030\a
reply \e]4;3;rgb:1111/2222/3333\a
reply \e]4;4;rgb:1212/4545/7878\a
reply \e]4;5;rgb:1212/5656/9a9a\a

## A color that cannot be read changes nothing
# Names of colors among them, which Twominal does not read.
size 10x1
send \e]4;1;red\a\e]4;1;rgb:1/2\a\e]4;1;rgb:12345/0/0\a\e]4;1;#12\a\e]4;1;rgb:1/2/3/4\a\e]4;1;?\a
reply \e]4;1;rgb:0101/0101/0101\a

## A number that is not one of the palette's ends OSC 4
size 10x1
send \e]4;1;#ff0000;256;#00ff00;2;?\a\e]4;x;?;1;?\a\e]4;1;?\a
reply \e]4;1;rgb:ffff/0000/0000\a

## OSC 104 gives colors of the palette back to the theme
size 10x1
send \e]4;1;#ff0000;2;#00ff00;3;#0000ff\a\e]104;1;3\a\e]4;1;?;2;?;3;?\a\e]104\a\e]4;2;?\a
reply \e]4;1;rgb:0101/0101/0101\a
reply \e]4;2;rgb:0000/ffff/0000\a
reply \e]4;3;rgb:0303/0303/0303\a
reply \e]4;2;rgb:0202/0202/0202\a

## OSC 110, 111 and 112 give the foreground, the background and the cursor back
size 10x1
send \e]10;#ffffff\a\e]11;#000000\a\e]12;#ff0000\a\e]110\a\e]11;?\a\e]111\a\e]112\a\e]10;?;?;?\a
reply \e]11;rgb:0000/0000/0000\a
reply \e]10;rgb:c0c0/c0c0/c0c0\a
reply \e]11;rgb:1010/1010/1010\a
reply \e]12;rgb:f0f0/f0f0/0000\a

## RIS and DECSTR give the palette back, and keep the other colors
# xterm's ReallyReset calls ResetAnsiColorRequest for both, and leaves the dynamic colors.
size 10x1
send \e]4;1;#ff0000\a\e]11;#000000\a\ec\e]4;1;?\a\e]11;?\a\e]4;1;#ff0000\a\e[!p\e]4;1;?\a
reply \e]4;1;rgb:0101/0101/0101\a
reply \e]11;rgb:0000/0000/0000\a
reply \e]4;1;rgb:0101/0101/0101\a

## A question cut short is not answered
size 10x1
send \e]11;?\e[c
reply \e[?62;22c
