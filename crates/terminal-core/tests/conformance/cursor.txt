# Moving, saving and restoring the cursor (TERM-06).

## CUP and HVP count from one and stay on the screen
size 20x10
send \e[3;5H
cursor 3;5
send \e[H
cursor 1;1
send \e[4;6f
cursor 4;6
send \e[0;0H
cursor 1;1
send \e[99;99H
cursor 10;20
send \e[65535;65535H
cursor 10;20

## A parameter too large for any screen still means the last row
size 20x10
send \e[99999999;3H
cursor 10;3

## Origin mode counts rows from the region and keeps the cursor in it
size 10x10
send \e[3;6r\e[?6h
cursor 3;1
send \e[2;3H
cursor 4;3
send \e[99H
cursor 6;1
send \e[99;99H
cursor 6;10
send \e[6n
reply \e[4;10R
send \e[2d
cursor 4;10
send \e[?6l
cursor 1;1

## DECRC restores the position and the character sets DECSC saved
size 10x5
send \e[3;4H\e(0\e7\e[H\e(B\e8q
screen
|          |
|          |
|   ─      |
|          |
|          |
cursor 3;5

## DECRC restores origin mode
size 10x5
send \e[2;4r\e[?6h\e7\e[?6l\e8\e[H
cursor 2;1

## DECRC restores a pending wrap
size 10x2
send 0123456789\e7\e[2;1H\e8x
screen
|0123456789|>
|x         |
cursor 2;2

## DECRC with nothing saved goes home
size 10x5
send \e[3;3H\e8
cursor 1;1

## CSI s and CSI u save and restore the cursor
size 10x5
send \e[2;3H\e[s\e[H\e[u
cursor 2;3

## With a private marker CSI u is another sequence
# XTRESTORE and the kitty keyboard protocol's, which are not done.
size 10x5
send \e[2;3H\e[s\e[4;4H
send \e[?u
cursor 4;4
send \e[?1u
cursor 4;4
send \e[=5u
cursor 4;4
send \e[>1u
cursor 4;4
send \e[<u
cursor 4;4

## Each screen has a saved cursor of its own

cursor 2;2
send \e[?47h\e[H\e8
cursor 4;4
