# Erasing, inserting and deleting characters (TERM-06). Each starts from three rows of four
# letters, with the cursor on the f, unless it says otherwise.

## ED erases from the cursor to the end of the screen
size 4x3
send abcd\r\nefgh\r\nijkl\e[2;2H\e[J
screen
|abcd|
|e   |
|    |
cursor 2;2

## ED 0 is ED
size 4x3
send abcd\r\nefgh\r\nijkl\e[2;2H\e[0J
screen
|abcd|
|e   |
|    |

## ED 1 erases from the start of the screen to the cursor, its cell included
size 4x3
send abcd\r\nefgh\r\nijkl\e[2;2H\e[1J
screen
|    |
|  gh|
|ijkl|
cursor 2;2

## ED 2 erases the whole screen and leaves the cursor
size 4x3
send abcd\r\nefgh\r\nijkl\e[2;2H\e[2J
screen
|    |
|    |
|    |
cursor 2;2

## ED 3 erases the scrollback, not the screen
# xterm's do_erase_display, with eraseSavedLines on as it is by default.
size 4x3
send 0123\r\nabcd\r\nefgh\r\nijkl\e[2;2H
scrollback 1
|0123|
send \e[3J
screen
|abcd|
|efgh|
|ijkl|
scrollback 0
cursor 2;2

## DECSED is done as ED
# Twominal has no protected characters, so selective erase erases everything.
size 4x3
send abcd\r\nefgh\r\nijkl\e[2;2H\e[?2J
screen
|    |
|    |
|    |

## Erasing to the end of a row ends its wrap
size 4x2
send abcdef\e[1;3H\e[K
screen
|ab  |
|ef  |

## Erasing the start of a row keeps its wrap
size 4x2
send abcdef\e[1;3H\e[1K
screen
|   d|>
|ef  |

## Erasing a row's last cell ends its wrap
# A deviation (DESIGN.md §3): xterm keeps the mark after EL 1 and EL 2, where this clears it
# once the erase reaches the last cell. The mark is only for copying and reflow, and an erased
# end of a row does not go on in the next.
size 4x2
send abcdef\e[1;4H\e[1K
screen
|    |
|ef  |
send \e[Habcdef\e[1;2H\e[2K
screen
|    |
|ef  |
send \e[Habcdef\e[1;2H\e[3X
screen
|a   |
|ef  |

## ECH erases characters from the cursor and does not move it
size 10x1
send abcdefgh\e[3G\e[2X
screen
|ab  efgh  |
cursor 1;3
send \e[5G\e[X
screen
|ab   fgh  |
cursor 1;5
send \e[3G\e[999X
screen
|ab        |
cursor 1;3

## ICH inserts blanks at the cursor and pushes the rest right
size 10x1
send abcdefgh\e[3G\e[2@
screen
|ab  cdefgh|
cursor 1;3
send \e[999@
screen
|ab        |
cursor 1;3

## DCH deletes characters at the cursor and pulls the rest left
size 10x1
send abcdefgh\e[3G\e[2P
screen
|abefgh    |
cursor 1;3
send \e[999P
screen
|ab        |
cursor 1;3

## DCH ends a row's wrap
# xterm's ScrnDeleteChar clears the mark.
size 4x2
send abcdef\e[1;2H\e[P
screen
|acd |
|ef  |

## ICH, DCH and ECH in the last column cancel a pending wrap
size 5x2
send abcde\e[@x
screen
|abcdx|
|     |
cursor 1;5
send \rabcde\e[Py
screen
|abcdy|
|     |
send \rabcde\e[Xz
screen
|abcdz|
|     |
cursor 1;5
