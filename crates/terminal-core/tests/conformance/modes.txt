# The modes a program sets, which the rest of the tab acts on (TERM-06, TERM-07, TERM-10,
# TERM-11).

## Modes are kept for the rest of the tab
size 10x2
send \e[?1h\e=\e[?2004h\e[?1004h\e[?1002h\e[?1006h\e[?25l\e[20h
mode application_cursor on
mode application_keypad on
mode bracketed_paste on
mode focus on
mode mouse drag
mode mouse_format sgr
mode cursor_visible off
mode newline on

## Turning off any mouse mode turns mouse reporting off
size 10x2
send \e[?1002h\e[?1000l
mode mouse off
send \e[?9;1003h
mode mouse motion
send \e[?9h
mode mouse press
send \e[?1000h
mode mouse click

## The mouse's formats exclude each other
# xterm's extend_coords: setting a format makes it the one in use, and resetting one that is
# not in use does nothing.
size 10x2
mode mouse_format legacy
send \e[?1006h
mode mouse_format sgr
send \e[?1015h
mode mouse_format urxvt
send \e[?1006l
mode mouse_format urxvt
send \e[?1015l
mode mouse_format legacy
send \e[?1005h
mode mouse_format utf8
send \e[?1006;1015h
mode mouse_format urxvt
send \e[?1015;1006h\e[?1006l
mode mouse_format legacy

## SGR-pixel reports are not done
# Twominal reports no pixel sizes, so a program could not tell cells by them (DESIGN.md §3).
# The format in use stays.
size 10x2
send \e[?1006h\e[?1016h
mode mouse_format sgr
send \e[?1016l
mode mouse_format sgr

## Turning tracking off keeps the format
size 10x2
send \e[?1002h\e[?1006h\e[?1002l
mode mouse off
mode mouse_format sgr

## Alternate scroll is on until a program turns it off
# xterm's is off until a program turns it on (DESIGN.md §3).
size 10x2
mode alternate_scroll on
send \e[?1007l
mode alternate_scroll off
send \e[?1007h
mode alternate_scroll on

## The keypad modes are set by ESC = and ESC >, and by mode 66
size 10x2
send \e=
mode application_keypad on
send \e>
mode application_keypad off
send \e[?66h
mode application_keypad on
send \e[?66l
mode application_keypad off

## Modes that would lock the keyboard, echo or resize are ignored
# KAM, SRM and DECCOLM (DESIGN.md §3).
size 10x2
send text\e[2h\e[12h\e[?3h\e[?3l
screen
|text      |
|          |
cursor 1;5

## An unknown mode is ignored
size 10x2
send \e[?99h\e[99h\e[?1;99h
mode application_cursor on

## DECSCUSR sets the cursor's shape and whether it blinks
# ctlseqs: CSI Ps SP q. The settings' shape is in use until a program asks for one.
size 10x2
mode cursor_style default
send \e[1 q
mode cursor_style blinking block
send \e[2 q
mode cursor_style steady block
send \e[3 q
mode cursor_style blinking underline
send \e[4 q
mode cursor_style steady underline
send \e[5 q
mode cursor_style blinking bar
send \e[6 q
mode cursor_style steady bar

## DECSCUSR 0, no parameter and 7 give the shape back to the settings
# xterm's 0 is a blinking block. Here it is the settings' shape, as in kitty, Ghostty,
# WezTerm, foot and Alacritty, and as xterm's 7, its own resources (DESIGN.md §3).
size 10x2
send \e[6 q\e[0 q
mode cursor_style default
send \e[4 q\e[ q
mode cursor_style default
send \e[2 q\e[7 q
mode cursor_style default

## DECSCUSR 8 and above, and sub-parameters, are ignored
size 10x2
send \e[6 q\e[8 q\e[65535 q\e[2:1 q
mode cursor_style steady bar

## The cursor's shape stays through DECSC, DECRC and the alternate screen
# xterm keeps one shape for both screens, and DECSC does not save it.
size 10x2
send \e7\e[5 q\e8
mode cursor_style blinking bar
send \e[?1049h\e[2 q
mode cursor_style steady block
send \e[?1049l
mode cursor_style steady block
send \e[4 q\e[?47h
mode cursor_style steady underline
